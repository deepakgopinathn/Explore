REPORT z_inline_declaration_program.

CLASS lcl_main DEFINITION.

  PUBLIC SECTION.
    TYPES: BEGIN OF ty_helper_type,
             vbeln TYPE vbak-vbeln,
             vkorg TYPE vbak-vkorg,
             vkgrp TYPE vbak-vkgrp,
           END OF ty_helper_type.
    CLASS-METHODS: main
      EXPORTING
        ls_vbak TYPE ty_helper_type.

ENDCLASS.

CLASS lcl_main IMPLEMENTATION.

  METHOD main.
*  data: lv_lines TYPE i.
    SELECT
     FROM vbak
     FIELDS vbeln,vkorg,vkgrp           //using FIELDS helps to get field list using Ctrl+1
     INTO TABLE @DATA(it_vbak)
     UP TO 100 ROWS.
    IF sy-subrc IS INITIAL.
      DATA(lv_lines) = lines( it_vbak  ).          //inline declaration
    ENDIF.

    LOOP AT it_vbak ASSIGNING FIELD-SYMBOL(<ls_vbak>).         //inline declaration

    ENDLOOP.

    READ TABLE it_vbak ASSIGNING FIELD-SYMBOL(<ls_vbak1>)
                       INDEX 1.
    READ TABLE it_vbak INTO ls_vbak INDEX 1.

    cl_salv_table=>factory(
*      EXPORTING
*        list_display   = if_salv_c_bool_sap=>false
*        r_container    =
*        container_name =
      IMPORTING
        r_salv_table   =  DATA(lr_alv)                      //inline declaration
      CHANGING
        t_table        = it_vbak
    ).
*    CATCH cx_salv_msg.
    lr_alv->display( ).


  ENDMETHOD.

ENDCLASS.




START-OF-SELECTION.

  lcl_main=>main( ).

  DATA(lcl_main) = NEW lcl_main( ).  //new syntax to create object
